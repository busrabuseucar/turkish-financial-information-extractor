import re
from typing import Callable, Optional

from .models import Evidence, FinancialRecord
from .normalization import normalize_date, normalize_maturity, normalize_number, normalize_percentage


class FinancialInformationExtractor:
    """Hybrid rule-based extractor for Turkish banking and campaign texts.

    The extractor intentionally returns evidence spans and confidence scores so
    every structured value can be audited against the original sentence.
    """

    PRODUCT_PATTERNS = [
        re.compile(r"(?P<value>[A-ZÇĞİÖŞÜ][\wÇĞİÖŞÜçğıöşü\- ]{2,50}(?:Finansmanı|Kredisi|Hesabı|Kartı))"),
        re.compile(r"(?P<value>[A-ZÇĞİÖŞÜ][\wÇĞİÖŞÜçğıöşü\- ]{2,50})\s+(?:kampanyası|ürünü)"),
    ]
    AMOUNT_PATTERNS = [
        re.compile(r"(?P<value>(?:en fazla|azami|kadar)?\s*[0-9][0-9.,\s]*\s*(?:TL|₺))", re.I),
    ]
    RATE_PATTERNS = [
        re.compile(r"(?P<value>%\s*[0-9]+(?:[.,][0-9]+)?|[0-9]+(?:[.,][0-9]+)?\s*%)\s*(?:kâr payı|kar payı|oran)", re.I),
        re.compile(r"(?:kâr payı|kar payı|oran)[^.%]{0,20}(?P<value>%\s*[0-9]+(?:[.,][0-9]+)?|[0-9]+(?:[.,][0-9]+)?\s*%)", re.I),
    ]
    MATURITY_PATTERNS = [
        re.compile(r"(?P<value>(?:en fazla|azami|kadar)?\s*\d+\s*(?:ay|yıl|gün))", re.I),
    ]
    FEE_PATTERNS = [
        re.compile(r"(?P<value>[0-9][0-9.,\s]*\s*(?:TL|₺))\s*(?:tahsis ücreti|masraf|ücret)", re.I),
        re.compile(r"(?:tahsis ücreti|masraf|ücret)[^0-9]{0,20}(?P<value>[0-9][0-9.,\s]*\s*(?:TL|₺))", re.I),
        re.compile(r"(?P<value>masrafsız|ücretsiz)", re.I),
    ]
    TARGET_PATTERNS = [
        re.compile(r"(?P<value>(?:18|21) yaşını doldurmuş[^.]{0,100})", re.I),
        re.compile(r"(?P<value>(?:emeklilere|öğrencilere|çiftçilere|KOBİ'lere|kobiler[e]?|yeni müşterilere|bireysel müşterilere))", re.I),
    ]
    DATE_PATTERNS = [
        re.compile(r"(?P<value>\d{1,2}[./-]\d{1,2}[./-]\d{4})"),
        re.compile(r"(?P<value>\d{1,2}\s+(?:Ocak|Şubat|Mart|Nisan|Mayıs|Haziran|Temmuz|Ağustos|Eylül|Ekim|Kasım|Aralık)\s+\d{4})", re.I),
    ]

    def _extract(
        self,
        text: str,
        patterns: list[re.Pattern],
        normalizer: Callable[[str], str],
        confidence: float,
    ) -> Optional[Evidence]:
        for pattern in patterns:
            match = pattern.search(text)
            if match:
                raw = match.group("value").strip()
                start, end = match.span("value")
                return Evidence(
                    value=normalizer(raw),
                    source_text=text[start:end],
                    start=start,
                    end=end,
                    confidence=confidence,
                )
        return None

    def extract(self, text: str) -> FinancialRecord:
        if not isinstance(text, str) or not text.strip():
            raise ValueError("text must be a non-empty string")

        return FinancialRecord(
            product_name=self._extract(text, self.PRODUCT_PATTERNS, str.strip, 0.82),
            financing_amount=self._extract(text, self.AMOUNT_PATTERNS, normalize_number, 0.90),
            profit_rate=self._extract(text, self.RATE_PATTERNS, normalize_percentage, 0.94),
            maturity=self._extract(text, self.MATURITY_PATTERNS, normalize_maturity, 0.91),
            fee=self._extract(text, self.FEE_PATTERNS, lambda x: "0" if x.lower() in {"masrafsız", "ücretsiz"} else normalize_number(x), 0.88),
            target_audience=self._extract(text, self.TARGET_PATTERNS, str.strip, 0.76),
            campaign_period=self._extract(text, self.DATE_PATTERNS, normalize_date, 0.87),
        )
